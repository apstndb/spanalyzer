package main

import (
	"cloud.google.com/go/spanner"
	database "cloud.google.com/go/spanner/admin/database/apiv1"
	dbpb "cloud.google.com/go/spanner/admin/database/apiv1/databasepb"
	sppb "cloud.google.com/go/spanner/apiv1/spannerpb"
	"context"
	"crypto/sha256"

	"errors"
	"fmt"

)

var ctx = context.Background()
var c *spanner.Client
var a *database.DatabaseAdminClient
var db string
var dialect string
var q = "ManagedQueueProbe20261029r1"

func emit(v map[string]any) {
	v["at"] = time.Now().UTC().Format(time.RFC3339Nano)
	v["dialect"] = dialect
	json.NewEncoder(os.Stdout).Encode(v)
}
func clean(e error) string {
	if e == nil {
		return ""
	}
	s := e.Error()
	s = strings.ReplaceAll(s, db, "<database>")
	for _, k := range []string{"SPANNER_PROJECT_ID", "SPANNER_INSTANCE_ID", "SPANNER_DATABASE_ID"} {
		if v := os.Getenv(k); v != "" {
			s = strings.ReplaceAll(s, v, "<redacted>")
		}
	}
	return s
}
func query(label, sql string, params map[string]any, stale bool, first bool) []map[string]any {
	start := time.Now()
	cx, cancel := context.WithTimeout(ctx, 35*time.Second)
	defer cancel()
	tx := c.Single()
	defer tx.Close()
	if stale {
		tx = tx.WithTimestampBound(spanner.ExactStaleness(time.Millisecond))
	}
	it := tx.Query(cx, spanner.Statement{SQL: sql, Params: params})
	defer it.Stop()
	rows := []map[string]any{}
	retained := []map[string]any{}
	types := map[string]string{}
	var err error
	for {
		var r *spanner.Row
		r, err = it.Next()
		if err != nil {
			break
		}
		row := map[string]any{}
		rr := map[string]any{}
		for i, n := range r.ColumnNames() {
			var g spanner.GenericColumnValue
			if err = r.Column(i, &g); err != nil {
				break
			}
			types[n] = g.Type.Code.String()
			v := g.Value.AsInterface()
			row[n] = v
			rr[n] = v
			if strings.Contains(n, "Token") && v != nil {
				rr[n] = fmt.Sprintf("sha256:%x", sha256.Sum256([]byte(fmt.Sprint(v))))
			}
		}
		if err != nil {
			break
		}
		rows = append(rows, row)
		retained = append(retained, rr)
		if first {
			if label == "receive-immediate" {
				if token, ok := row["SpannerLeaseToken"].(string); ok {
					leaseControls(token)
				}
			}
			break
		}
	}
	if err == iterator.Done {
		err = nil
	}
	emit(map[string]any{"kind": "query", "label": label, "sql": sql, "stale": stale, "stop_after_first": first, "rows": retained, "column_types": types, "error": clean(err), "code": status.Code(err).String(), "elapsed_ms": time.Since(start).Milliseconds()})
	return rows
}
func tx(label string, sqls []string, rollback bool) {
	var counts []int64
	cx, cancel := context.WithTimeout(ctx, 35*time.Second)
	defer cancel()
	_, err := c.ReadWriteTransaction(cx, func(cx context.Context, t *spanner.ReadWriteTransaction) error {
		counts = nil
		for _, sql := range sqls {
			n, e := t.Update(cx, spanner.Statement{SQL: sql})
			counts = append(counts, n)
			if e != nil {
				return e
			}
		}
		if rollback {
			return errors.New("intentional probe rollback")
		}
		return nil
	})
	emit(map[string]any{"kind": "transaction", "label": label, "sql": sqls, "rows_modified": counts, "intentional_rollback": rollback, "error": clean(err), "code": status.Code(err).String()})
}
func ddl(label, sql string) bool {
	cx, cancel := context.WithTimeout(ctx, 4*time.Minute)
	defer cancel()
	op, e := a.UpdateDatabaseDdl(cx, &dbpb.UpdateDatabaseDdlRequest{Database: db, Statements: []string{sql}})
	if e == nil {
		e = op.Wait(cx)
	}
	emit(map[string]any{"kind": "ddl", "label": label, "sql": sql, "error": clean(e), "code": status.Code(e).String()})
	return e == nil
}

func observe(label, sql string, params map[string]any, mode sppb.ExecuteSqlRequest_QueryMode) {
	cx, cancel := context.WithTimeout(ctx, 25*time.Second)
	defer cancel()
	ro := c.Single()
	defer ro.Close()
	it := ro.QueryWithOptions(cx, spanner.Statement{SQL: sql, Params: params}, spanner.QueryOptions{Mode: &mode})
	defer it.Stop()
	count := 0
	var e error
	start := time.Now()
	for {
		_, e = it.Next()
		if e != nil {
			break
		}
		count++
	}
	if e == iterator.Done {
		e = nil
	}
	v := map[string]any{"kind": "plan-probe", "label": label, "sql": sql, "params": params, "mode": mode.String(), "error": clean(e), "code": status.Code(e).String(), "rows": count, "elapsed_ms": time.Since(start).Milliseconds(), "stats": it.QueryStats}
	if it.QueryPlan != nil {
		b, _ := protojson.Marshal(it.QueryPlan)
		v["plan"] = json.RawMessage(b)
	}
	if it.Metadata != nil {
		b, _ := protojson.Marshal(it.Metadata)
		v["metadata"] = json.RawMessage(b)
	}
	emit(v)
}

func leaseControls(token string) {
	query("control-scalar-default", "SELECT * FROM RENEWLEASE_"+q+"(lease_tokens=>[@token])", map[string]any{"token": token}, false, false)
	query("control-scalar-off", "@{AUTO_PARAMETER='OFF'} SELECT * FROM RENEWLEASE_"+q+"(lease_tokens=>[@token])", map[string]any{"token": token}, false, false)
	rows := query("control-array-default", "SELECT * FROM RENEWLEASE_"+q+"(lease_tokens=>@tokens)", map[string]any{"tokens": []string{token}}, false, false)
	if len(rows) > 0 {
		if next, ok := rows[0]["SpannerNewLeaseToken"].(string); ok {
			token = next
		}
	}
	query("control-array-off", "@{AUTO_PARAMETER='OFF'} SELECT * FROM RENEWLEASE_"+q+"(lease_tokens=>@tokens)", map[string]any{"tokens": []string{token}}, false, false)
}
func main() {
	if os.Getenv("SPANNER_EMULATOR_HOST") != "" {
		panic("emulator override")
	}
	db = os.Getenv("TEST_REAL_SPANNER_DATABASE")
	q = "QueueRecreate20261029"
	dialect = "GOOGLE_STANDARD_SQL"
	var e error
	a, e = database.NewDatabaseAdminClient(ctx)
	if e != nil {
		panic(clean(e))
	}
	defer a.Close()
	c, e = spanner.NewClient(ctx, db)
	if e != nil {
		panic(clean(e))
	}
	defer c.Close()
	query("preflight", "SELECT 1 AS ready", nil, false, false)
	owned := false
	defer func() {
		if owned && ddl("cleanup", "DROP QUEUE "+q) {
			query("cleanup-confirmation", "SELECT COUNT(*) AS remaining FROM INFORMATION_SCHEMA.TABLES WHERE TABLE_NAME='"+q+"'", nil, false, false)
		}
	}()
	for cycle := 1; cycle <= 2; cycle++ {
		label := fmt.Sprintf("cycle-%d", cycle)
		if !ddl(label+"-create", "CREATE QUEUE "+q+" (MessageId STRING(36) NOT NULL, Payload BYTES(MAX) NOT NULL) PRIMARY KEY(MessageId)") {
			return
		}
		owned = true
		id := fmt.Sprintf("generation-%d", cycle)
		tx(label+"-enqueue", []string{"INSERT INTO " + q + " (MessageId,Payload,DeliverTime) VALUES ('" + id + "',b'probe',CURRENT_TIMESTAMP())"}, false)
		rows := query(label+"-receive", "SELECT * FROM RECEIVE_"+q+"(max_duration=>'3s', max_batch_size=>1)", nil, false, true)
		expected := len(rows) == 1 && rows[0]["MessageId"] == id
		emit(map[string]any{"label": label + "-expected-generation", "matches": expected, "expected_message_id": id, "row_count": len(rows)})
		if !expected {
			return
		}
		if cycle == 1 {
			if !ddl(label+"-drop", "DROP QUEUE "+q) {
				return
			}
			owned = false
		}
	}
}
