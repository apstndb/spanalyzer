"""Run under the dedicated user-level mise profile; retain redacted output only."""
from pathlib import Path
import datetime, json, os, re, subprocess, sys, time

root = Path('<LOCAL_CHECKOUT>')
out = root / '.tmp/managed-recheck-20261029'
durable = Path('<LOCAL_TASK_ROOT>/apstndb/spanalyzer/managed-recheck-20261029')
go = '<LOCAL_HOME>/.local/share/mise/installs/go/1.26.8/bin/go'
env = os.environ.copy()
for key in ['SPANNER_PROJECT_ID', 'SPANNER_INSTANCE_ID', 'SPANNER_DATABASE_ID']:
    if not env.get(key):
        raise RuntimeError('Dedicated profile missing ' + key)
if env.get('SPANNER_EMULATOR_HOST'):
    raise RuntimeError('Emulator override is present')
env['TEST_REAL_SPANNER_DATABASE'] = 'projects/{SPANNER_PROJECT_ID}/instances/{SPANNER_INSTANCE_ID}/databases/{SPANNER_DATABASE_ID}'.format(**env)
for key in list(env):
    if key.startswith('SPANEMUBOOST_') or key.startswith('REQUIRE_REAL_SPANNER_'):
        env.pop(key, None)
env.update(GOTOOLCHAIN='local', GOROOT=str(Path(go).parent.parent), GOFLAGS='-mod=readonly')
env['PATH'] = str(Path(go).parent) + ':' + env['PATH']
env['GOCACHE'] = str(root / '.tmp/managed-recheck-20260925/go-cache')
env.update({k:'1' for k in ['REQUIRE_REAL_SPANNER_CANONICAL','REQUIRE_REAL_SPANNER_DRIFT','REQUIRE_REAL_SPANNER_EXPRESSION_INDEX','REQUIRE_REAL_SPANNER_UUID']})
secrets = [env[k] for k in ['TEST_REAL_SPANNER_DATABASE','SPANNER_PROJECT_ID','SPANNER_INSTANCE_ID','SPANNER_DATABASE_ID']]
receipts_path = durable/'evidence/post-graph-final-commands.json'
receipts = json.loads(receipts_path.read_text()) if receipts_path.exists() else []

def scrub(value):
    if isinstance(value,str):
        for secret in secrets:
            value=value.replace(secret,'<redacted>')
        value=re.sub(r'requestID = "[^"]*"','requestID = "<redacted>"',value)
        return re.sub(r'projects/<redacted>/instances/<redacted>/databases/[A-Za-z0-9_-]+','<database>',value)
    if isinstance(value,list):return [scrub(x) for x in value]
    if isinstance(value,dict):return {k:scrub(x) for k,x in value.items()}
    return value

def redact(text):
    lines=[]
    for line in text.splitlines():
        try:line=json.dumps(scrub(json.loads(line)),ensure_ascii=False)
        except ValueError:line=scrub(line)
        lines.append(line)
    return '\n'.join(lines)+('\n' if lines else '')

def run(name, args, cwd, workspace=False, timeout=1200):
    previous = next((x for x in receipts if x['name']==name),None)
    if previous:
        print('RETAINED '+name+' exit='+str(previous['exit_code']),flush=True)
        return previous['exit_code']
    case_env = env.copy()
    case_env['GOWORK'] = str(root/'go.work') if workspace else 'off'
    print('START ' + name, flush=True)
    at = datetime.datetime.now(datetime.timezone.utc).isoformat()
    started = time.time()
    proc = subprocess.run(args, cwd=cwd, env=case_env, capture_output=True, text=True, timeout=timeout)
    for stream in ['stdout','stderr']:
        text = redact(getattr(proc,stream))
        (out/(name+'.'+stream)).write_text(text)
        (durable/'evidence'/(name+'.'+stream)).write_text(text)
    entry = {'name':name,'started_at':at,'completed_at':datetime.datetime.now(datetime.timezone.utc).isoformat(),'exit_code':proc.returncode,'seconds':round(time.time()-started,3),'argv':args,'cwd':str(cwd),'gowork':case_env['GOWORK']}
    receipts.append(entry)
    (durable/'evidence/post-graph-final-commands.json').write_text(json.dumps(receipts,indent=2)+'\n')
    print(json.dumps(entry),flush=True)
    return proc.returncode


wait_started=time.time()
while True:
    path=durable/'evidence/graph-final-commands.json'
    prior=json.loads(path.read_text())if path.exists()else[]
    if any(r['name']=='graph-final'and r['exit_code']==0 for r in prior):break
    if time.time()-wait_started>1800:raise SystemExit('Repair completion unavailable')
    time.sleep(5)
for name,args in [('post-graph-inventory',[str(out/'query'),str(out/'final-inventory-cases.json')]),('post-graph-trial-state',[sys.executable,str(out/'inspect-cloud-state.py')])]:
    if run(name,args,root/'tools',workspace=True):raise SystemExit('Final read-only verification failed')
print('COMPLETE: final private target inventory recorded',flush=True)
