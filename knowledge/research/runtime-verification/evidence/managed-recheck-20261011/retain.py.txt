def normalized(value):
    text = json.dumps(value,sort_keys=True)
    text = re.sub(r'202610(?:09|11)r1', '<fixture-date>',text)
    text = re.sub(r'docdiff-202610(?:10|11)[gp]\d+', '<temporary-database>',text)
    return text

def paired(before,after,values=False):
    def key(x): return (x.get('label'),x.get('kind'),normalized(x.get('sql')))
    left,right=collections.defaultdict(list),collections.defaultdict(list)
    for x in before:left[key(x)].append(x)
    for x in after:right[key(x)].append(x)
    changes=[];rowchanges=[];typechanges=[];n=0
    for key,rows in left.items():
        for a,b in zip(rows,right.get(key,[])):
            n+=1
            if a.get('code')!=b.get('code'):changes.append({'label':a.get('label'),'sql':b.get('sql'),'before':a.get('code'),'after':b.get('code')})
            if values and normalized(a.get('rows'))!=normalized(b.get('rows')):rowchanges.append(b.get('label'))
            if values and a.get('metadata',{}).get('rowType')!=b.get('metadata',{}).get('rowType'):typechanges.append(b.get('label'))
    return {'baseline_records':len(before),'current_records':len(after),'comparable_records':n,'changed_codes':changes,'changed_values':rowchanges if values else None,'changed_row_types':typechanges if values else None,'value_comparison':'deterministic rows and result row types' if values else 'not compared; workload, heartbeat timestamps, sampled rows and plans can vary'}

def walk(value):
    if isinstance(value,dict):
        yield value
        for x in value.values():yield from walk(x)
    elif isinstance(value,list):
        for x in value:yield from walk(x)
