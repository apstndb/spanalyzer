cases, expected = [], {}
for name, left, right, value in pairs:
    cases.append({'label': name, 'sql': f'SELECT ({left}) IS DISTINCT FROM ({right}) AS d, ({left}) IS NOT DISTINCT FROM ({right}) AS nd', 'params': None, 'profile': False})
    if value is not None:
        expected[name] = {'code': 'OK', 'rows': [{'d': value, 'nd': not value}]}
for name, sql in [
    ('equality-nan-control', "SELECT CAST('NaN' AS FLOAT64) = CAST('NaN' AS FLOAT64) AS eq"),

    ('missing-distinct-control', 'SELECT 1 IS NOT FROM 1 AS result'),
]:

expected['equality-null-control'] = {'code': 'OK', 'rows': [{'eq': None}]}
for name in ['missing-from-control', 'missing-rhs-control', 'missing-distinct-control']:
    expected[name] = {'code': 'InvalidArgument'}
for name in ['int-equal', 'nan-nan', 'sql-null-json-null', 'json-object-order', 'array-null-elements']:
    case = next(x for x in cases if x['label'] == name).copy()
    case.update(label=name+'-profile', profile=True)
    cases.append(case)
    if name in expected:
        expected[case['label']] = expected[name]
(task / 'distinctness-cases.json').write_text(json.dumps(cases, indent=2)+'\n')
(task / 'distinctness-expected.json').write_text(json.dumps(expected, indent=2)+'\n')
