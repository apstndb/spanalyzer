func findTokens(v any) []string {
    result := []string{}
    switch x := v.(type) {
    case map[string]any:
        if token, ok := x["token"].(string); ok { result = append(result, token) }
        for key, value := range x { if key != "token" { result = append(result, findTokens(value)...)} }
    case []any:
        for _, value := range x { result = append(result, findTokens(value)...)}
    }
    return result
}
