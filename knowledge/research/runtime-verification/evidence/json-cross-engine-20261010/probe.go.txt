var ctx = context.Background()
var c *spanner.Client
var a *database.DatabaseAdminClient
var db string
var pg bool
var backend string
var phase string

var table = "DocDiffNodes"
var stream = "DocDiffStream"

func emit(v map[string]any) {
	v["backend"] = backend
	v["phase"] = phase
	v = scrub(v).(map[string]any)
	v["at"] = time.Now().UTC().Format(time.RFC3339Nano)
	json.NewEncoder(os.Stdout).Encode(v)
}
func outcome(label string, e error) {
	s := ""
	if e != nil {
		s = e.Error()
	}
	emit(map[string]any{"label": label, "code": status.Code(e).String(), "error": s})
}
func ddl(label string, sqls ...string) bool {
	cx, cancel := context.WithTimeout(ctx, 3*time.Minute)
	defer cancel()
	op, e := a.UpdateDatabaseDdl(cx, &dbpb.UpdateDatabaseDdlRequest{Database: db, Statements: sqls})
	if e == nil {
		e = op.Wait(cx)
	}
	s := ""
	if e != nil {
		s = e.Error()
	}
	emit(map[string]any{"label": label, "sql": sqls, "code": status.Code(e).String(), "error": s})
	return e == nil
}
func decode(t *sppb.Type, v any) any {
	if v == nil {
		return nil
	}
	if t.Code == sppb.TypeCode_ARRAY {
		xs := v.([]any)
		out := make([]any, len(xs))
		for i, x := range xs {
			out[i] = decode(t.ArrayElementType, x)
		}
		return out
	}
	if t.Code == sppb.TypeCode_STRUCT {
		xs := v.([]any)
		out := map[string]any{}
		for i, f := range t.StructType.Fields {
			out[f.Name] = decode(f.Type, xs[i])
		}
		return out
	}
	return v
}
func findToken(v any) string {
	switch x := v.(type) {
	case map[string]any:
		if t, ok := x["token"].(string); ok {
			return t
		}
		for _, value := range x {
			if s := findToken(value); s != "" {
				return s
			}
		}
	case []any:
		for _, value := range x {
			if s := findToken(value); s != "" {
				return s
			}
		}
	}
	return ""
}
func redactTokens(v any) any {
	switch x := v.(type) {
	case map[string]any:
		out := map[string]any{}
		for k, value := range x {
			if strings.Contains(strings.ToLower(k), "token") && value != nil {
				out[k] = fmt.Sprintf("sha256:%x", sha256.Sum256([]byte(fmt.Sprint(value))))
			} else {
				out[k] = redactTokens(value)
			}
		}
		return out
	case []any:
		out := make([]any, len(x))
		for i, value := range x {
			out[i] = redactTokens(value)
		}
		return out
	default:
		return v
	}
}
func query(label, sql string, params map[string]any, profile bool) []map[string]any {
	cx, cancel := context.WithTimeout(ctx, 30*time.Second)
	defer cancel()
	ro := c.Single()
	defer ro.Close()
	mode := sppb.ExecuteSqlRequest_NORMAL
	if profile {
		mode = sppb.ExecuteSqlRequest_PROFILE
	}
	it := ro.QueryWithOptions(cx, spanner.Statement{SQL: sql, Params: params}, spanner.QueryOptions{Mode: &mode})
	defer it.Stop()
	rows := []map[string]any{}
	var e error
	for {
		var row *spanner.Row
		row, e = it.Next()
		if e != nil {
			break
		}
		m := map[string]any{}
		for i, n := range row.ColumnNames() {
			var g spanner.GenericColumnValue
			e = row.Column(i, &g)
			if e != nil {
				break
			}
			m[n] = decode(g.Type, g.Value.AsInterface())
		}
		if e != nil {
			break
		}
		rows = append(rows, m)
	}
	if e == iterator.Done {
		e = nil
	}
	s := ""
	if e != nil {
		s = e.Error()
	}
	out := map[string]any{"label": label, "sql": sql, "params": params, "mode": mode.String(), "rows": rows, "code": status.Code(e).String(), "error": s}
	if it.Metadata != nil {
		b, _ := protojson.Marshal(it.Metadata)
		out["metadata"] = json.RawMessage(b)
	}
	if it.QueryPlan != nil {
		b, _ := protojson.Marshal(it.QueryPlan)
		out["plan"] = json.RawMessage(b)
	}
	if it.QueryStats != nil {
		out["stats"] = it.QueryStats
	}
	emit(out)
	return rows
}
func read(label string, start, end time.Time, token *string, hb int64, options string) string {
	sql := fmt.Sprintf("SELECT ChangeRecord FROM READ_%s(@start,@end,@token,%d%s)", stream, hb, options)
	if pg {
		sql = fmt.Sprintf("SELECT * FROM spanner.\"read_json_%s\"($1,$2,$3,%d%s)", stream, hb, options)
	}
	params := map[string]any{"start": start, "end": end, "token": spanner.NullString{}}
	if token != nil {
		params["token"] = *token
	}
	if pg {
		params = map[string]any{"p1": params["start"], "p2": params["end"], "p3": params["token"]}
	}
	cx, cancel := context.WithTimeout(ctx, 15*time.Second)
	defer cancel()
	ro := c.Single()
	defer ro.Close()
	it := ro.Query(cx, spanner.Statement{SQL: sql, Params: params})
	defer it.Stop()
	rows := []map[string]any{}
	var e error
	found := ""
	for {
		var r *spanner.Row
		r, e = it.Next()
		if e != nil {
			break
		}
		record := map[string]any{}
		for i, n := range r.ColumnNames() {
			var g spanner.GenericColumnValue
			if e = r.Column(i, &g); e != nil {
				break
			}
			v := decode(g.Type, g.Value.AsInterface())
			if pg {
				if s, ok := v.(string); ok {
					var obj any
					if json.Unmarshal([]byte(s), &obj) == nil {
						v = obj
					}
				}
			}
			record[n] = v
		}
		if e != nil {
			break
		}
		if found == "" {
			found = findToken(record)
		}
		rows = append(rows, map[string]any{"received_at": time.Now().UTC().Format(time.RFC3339Nano), "record": redactTokens(record)})
	}
	if e == iterator.Done {
		e = nil
	}
	s := ""
	if e != nil {
		s = e.Error()
	}
	emit(map[string]any{"label": label, "sql": sql, "start": start, "end": end, "partition": "root", "child_partition": token != nil, "heartbeat_ms": hb, "rows": rows, "code": status.Code(e).String(), "error": s})
	return found
}
func inspectScope(label string) {
	for _, view := range []string{"CHANGE_STREAMS", "CHANGE_STREAM_TABLES", "CHANGE_STREAM_COLUMNS", "CHANGE_STREAM_OPTIONS"} {
		query(label+"-"+view, "SELECT * FROM INFORMATION_SCHEMA."+view+" WHERE CHANGE_STREAM_NAME='"+stream+"'", nil, false)
	}
}
func descriptor() []byte {
	f := &descriptorpb.FileDescriptorProto{Name: proto.String("docdiff.proto"), Package: proto.String("docdiff"), Syntax: proto.String("proto2"), EnumType: []*descriptorpb.EnumDescriptorProto{{Name: proto.String("Color"), Value: []*descriptorpb.EnumValueDescriptorProto{{Name: proto.String("NEGATIVE"), Number: proto.Int32(-1)}, {Name: proto.String("ZERO"), Number: proto.Int32(0)}, {Name: proto.String("ONE"), Number: proto.Int32(1)}}}}, MessageType: []*descriptorpb.DescriptorProto{{Name: proto.String("Item"), Field: []*descriptorpb.FieldDescriptorProto{{Name: proto.String("value"), Number: proto.Int32(1), Label: descriptorpb.FieldDescriptorProto_LABEL_OPTIONAL.Enum(), Type: descriptorpb.FieldDescriptorProto_TYPE_INT64.Enum()}, {Name: proto.String("name"), Number: proto.Int32(2), Label: descriptorpb.FieldDescriptorProto_LABEL_OPTIONAL.Enum(), Type: descriptorpb.FieldDescriptorProto_TYPE_STRING.Enum()}}}}}
	b, e := proto.Marshal(&descriptorpb.FileDescriptorSet{File: []*descriptorpb.FileDescriptorProto{f}})
	if e != nil {
		panic(e)
	}
	return b
}

// Scrub locally generated resource identities too; image identities are retained.
func scrub(v any) any {
	switch x := v.(type) {
	case string:
		if db != "" {
			x = strings.ReplaceAll(x, db, "<database>")
			for _, part := range strings.Split(db, "/") {
				if part != "" && part != "projects" && part != "instances" && part != "databases" {
					x = strings.ReplaceAll(x, part, "<redacted>")
				}
			}
		}
		x = regexp.MustCompile(`projects/[^/\s]+/instances/[^/\s]+/databases/[^/\s\"]+`).ReplaceAllString(x, "<database>")
		x = regexp.MustCompile(`requestID = "[^"]*"`).ReplaceAllString(x, `requestID = "<redacted>"`)
		return x
	case []any:
		out := make([]any, len(x))
		for i, v := range x {
			out[i] = scrub(v)
		}
		return out
	case []map[string]any:
		out := make([]map[string]any, len(x))
		for i, v := range x {
			out[i] = scrub(v).(map[string]any)
		}
		return out
	case map[string]any:
		out := map[string]any{}
		for k, v := range x {
			out[k] = scrub(v)
		}
		return out
	default:
		return v
	}
}
func setup(rt spanemuboost.Runtime, d dbpb.DatabaseDialect) *spanemuboost.Clients {
	opts := []spanemuboost.Option{spanemuboost.WithRandomDatabaseID(), spanemuboost.ForceSchemaTeardown(), spanemuboost.WithDatabaseDialect(d)}
	if backend == "emulator" {
		opts = append(opts, spanemuboost.EnableAutoConfig())
	} else {
		opts = append(opts, spanemuboost.EnableDatabaseAutoConfigOnly())
	}
	clients, e := spanemuboost.OpenClients(ctx, rt, opts...)
	outcome("database-setup", e)
	if e != nil {
		return nil
	}
	c = clients.Client
	a = clients.DatabaseClient
	db = clients.DatabasePath()
	return clients
}
func finish(rt spanemuboost.Runtime, clients *spanemuboost.Clients) {
	outcome("database-cleanup", clients.Close())
	inspector, e := database.NewDatabaseAdminClient(ctx, rt.ClientOptions()...)
	if e != nil {
		outcome("cleanup-inspector", e)
		return
	}
	defer inspector.Close()
	_, e = inspector.GetDatabase(ctx, &dbpb.GetDatabaseRequest{Name: db})
	outcome("database-absent", e)
}
func constants() {
	for _, name := range []string{"cases.json", "followup-cases.json"} {
		f, e := os.Open(name)
		if e != nil {
			panic(e)
		}
		var cases []struct {
			Label   string         `json:"label"`
			SQL     string         `json:"sql"`
			Params  map[string]any `json:"params"`
			Profile bool           `json:"profile"`
		}
		e = json.NewDecoder(f).Decode(&cases)
		f.Close()
		if e != nil {
			panic(e)
		}
		for _, p := range cases {
			query(p.Label, p.SQL, p.Params, p.Profile)
		}
	}
}
func fixtures() {
	if pg {
		table = "docdiffnodes"
		stream = "docdiffstream"
	} else {
		table = "DocDiffNodes"
		stream = "DocDiffStream"
	}
	sql := "CREATE TABLE " + table + " (Id INT64 NOT NULL, Value INT64, Other INT64, J JSON) PRIMARY KEY(Id)"
	if pg {
		sql = "CREATE TABLE " + table + " (Id bigint PRIMARY KEY, Value bigint, Other bigint)"
	}
	if !ddl("create-table", sql) {
		return
	}
	graphReady := false
	if !pg {
		if ddl("create-edge-table", "CREATE TABLE DocDiffEdges (Src INT64 NOT NULL, Dst INT64 NOT NULL) PRIMARY KEY(Src,Dst)") {
			graphReady = ddl("create-graph", "CREATE PROPERTY GRAPH DocDiffGraph NODE TABLES ("+table+" KEY(Id) LABEL Node PROPERTIES(Id,Value)) EDGE TABLES (DocDiffEdges KEY(Src,Dst) SOURCE KEY(Src) REFERENCES "+table+"(Id) DESTINATION KEY(Dst) REFERENCES "+table+"(Id) LABEL Edge PROPERTIES(Src,Dst))")
		}
	}
	streamReady := ddl("create-stream", "CREATE CHANGE STREAM "+stream+" FOR "+table+"(Value)")
	var mutations []*spanner.Mutation
	for i := int64(1); i <= 4; i++ {
		cols := []string{"Id", "Value", "Other"}
		values := []any{i, i * 10, i * 100}
		if pg {
			cols = []string{"id", "value", "other"}
		} else {
			cols = append(cols, "J")
			values = append(values, spanner.NullJSON{Value: map[string]any{"n": i % 2}, Valid: true})
		}
		mutations = append(mutations, spanner.Insert(table, cols, values))
	}
	if graphReady {
		mutations = append(mutations, spanner.Insert("DocDiffEdges", []string{"Src", "Dst"}, []any{int64(1), int64(2)}))
	}
	_, e := c.Apply(ctx, mutations)
	outcome("seed", e)
	if e != nil {
		return
	}
	if streamReady {
		inspectScope("initial")
		for _, setting := range []string{"exclude_ttl_deletes=true", "exclude_ttl_deletes=false", "allow_txn_exclusion=true", "allow_txn_exclusion=false"} {
			clause := " SET OPTIONS (" + setting + ")"
			if pg {
				clause = " SET (" + setting + ")"
			}
			ddl("option-only-"+setting, "ALTER CHANGE STREAM "+stream+clause)
			inspectScope("after-" + setting)
		}
		clause := " SET OPTIONS (exclude_ttl_deletes=true)"
		if pg {
			clause = " SET (exclude_ttl_deletes=true)"
		}
		ddl("old-for-all-control", "ALTER CHANGE STREAM "+stream+" FOR ALL"+clause)
		now := time.Now().UTC()
		token := ""
		for _, hb := range []int64{99, 100, 999, 1000, 300000, 300001} {
			t := read(fmt.Sprintf("heartbeat-boundary-%d", hb), now, now.Add(time.Second), nil, hb, ",NULL")
			if t != "" {
				token = t
			}
		}
		read("read-options-omitted", now, now.Add(time.Second), nil, 100, "")
		options := ",ARRAY<STRING>[]"
		if pg {
			options = ",ARRAY[]::text[]"
		}
		read("read-options-empty", now, now.Add(time.Second), nil, 100, options)
		options = ",['unknown']"
		if pg {
			options = ",ARRAY['unknown']::text[]"
		}
		read("read-options-nonnull", now, now.Add(time.Second), nil, 100, options)
		if token != "" {
			now = time.Now().UTC()
			read("heartbeat-child-100", now, now.Add(2*time.Second), &token, 100, ",NULL")
		} else {
			emit(map[string]any{"label": "heartbeat-child-100", "status": "not-run", "reason": "No root query yielded a child partition token"})
		}
	}
	if pg {
		return
	}
	for label, sql := range map[string]string{
		"stored-json-sort":                      "SELECT Id,J FROM " + table + " ORDER BY J,Id",
		"stored-json-group":                     "SELECT J,COUNT(*) AS n FROM " + table + " GROUP BY J ORDER BY J",
		"conditional-profile":                   "SELECT COALESCE((SELECT Value FROM " + table + " WHERE Id=1),(SELECT MAX(Value) FROM " + table + " WHERE Id>=2),(SELECT SUM(Other) FROM " + table + ")) AS result",
		"conditional-profile-error-suppression": "SELECT COALESCE((SELECT Value FROM " + table + " WHERE Id=1),(SELECT MAX(Value/0) FROM " + table + " WHERE Id>=2)) AS result",
	} {
		query(label, sql, nil, true)
	}
	if graphReady {
		for _, p := range []struct{ label, sql string }{
			{"graph-node-control", "SELECT * FROM GRAPH_TABLE(DocDiffGraph MATCH (n) RETURN n.Id AS id)"},
			{"graph-edge-control", "SELECT * FROM GRAPH_TABLE(DocDiffGraph MATCH p=(n)-[e]->(m) RETURN n.Id AS src,m.Id AS dst)"},
			{"graph-element-cast-normal", "SELECT * FROM GRAPH_TABLE(DocDiffGraph MATCH (n) RETURN CAST(n AS JSON) AS j)"},
			{"graph-path-cast-normal", "SELECT * FROM GRAPH_TABLE(DocDiffGraph MATCH p=(n)-[e]->(m) RETURN CAST(p AS JSON) AS j)"},
			{"graph-element-to-json-control", "SELECT * FROM GRAPH_TABLE(DocDiffGraph MATCH (n) RETURN TO_JSON(n) AS j)"},
			{"graph-path-to-json-control", "SELECT * FROM GRAPH_TABLE(DocDiffGraph MATCH p=(n)-[e]->(m) RETURN TO_JSON(p) AS j)"},
			{"graph-property-cast-control", "SELECT * FROM GRAPH_TABLE(DocDiffGraph MATCH (n) RETURN CAST(n.Id AS JSON) AS j)"},
		} {
			query(p.label, p.sql, nil, false)
		}
	}
	op, e := a.UpdateDatabaseDdl(ctx, &dbpb.UpdateDatabaseDdlRequest{Database: db, Statements: []string{"CREATE PROTO BUNDLE (docdiff.Color,docdiff.Item)"}, ProtoDescriptors: descriptor()})
	if e == nil {
		e = op.Wait(ctx)
	}
	outcome("proto-bundle-setup", e)
	if e == nil {
		for label, expr := range map[string]string{
			"cast-enum":               "CAST(CAST('ONE' AS docdiff.Color) AS JSON)",
			"cast-proto":              "CAST(NEW docdiff.Item(1 AS value,'x' AS name) AS JSON)",
			"enum-equality":           "JSON '\"ONE\"' = CAST('ONE' AS docdiff.Color)",
			"enum-order-reject":       "JSON '\"ONE\"' < CAST('ZERO' AS docdiff.Color)",
			"enum-array-equality":     "JSON '[\"ONE\"]' = [CAST('ONE' AS docdiff.Color)]",
			"enum-array-order-reject": "JSON '[\"ONE\"]' < [CAST('ZERO' AS docdiff.Color)]",
		} {
			query(label, "SELECT "+expr+" AS result", nil, false)
		}
	}
}
func main() {
	backend = os.Args[1]
	image := os.Args[2]
	phase = "bootstrap"
	kind := spanemuboost.BackendOmni
	if backend == "emulator" {
		kind = spanemuboost.BackendEmulator
	}
	rt, e := spanemuboost.Run(ctx, kind, spanemuboost.WithContainerImage(image), spanemuboost.DisableAutoConfig())
	outcome("runtime-start", e)
	if e != nil {
		os.Exit(1)
	}
	defer func() { phase = "cleanup"; outcome("runtime-cleanup", rt.Close()) }()
	platform, e := spanemuboost.RuntimePlatform(ctx, rt)
	if e != nil {
		panic(e)
	}
	emit(map[string]any{"label": "identity", "image": image, "platform": platform})
	pg = false
	phase = "google-constants"
	clients := setup(rt, dbpb.DatabaseDialect_GOOGLE_STANDARD_SQL)
	if clients == nil {
		return
	}
	constants()
	phase = "google-fixtures"
	fixtures()
	finish(rt, clients)
	pg = true
	phase = "pg-fixtures"
	clients = setup(rt, dbpb.DatabaseDialect_POSTGRESQL)
	if clients == nil {
		return
	}
	fixtures()
	finish(rt, clients)
}
