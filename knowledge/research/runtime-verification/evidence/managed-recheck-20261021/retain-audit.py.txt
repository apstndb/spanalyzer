def group(path):
    p=path.removeprefix('docs/docs.cloud.google.com/')
    if p.startswith('spanner-omni'):return 'omni_documentation_context_only'
    if '/samples/' in p:return 'sample_code_repairs'
    if 'full-text-search/' in p or p.endswith(('search_functions.md','working-with-json.md','working-with-jsonb.md')):return 'json_search_equality'
    if p.endswith('graph-query-statements.md'):return 'gql_column_propagation'
    if p.endswith('manage-schemaless-data.md'):return 'dynamic_json_properties'
    if '/graph/' in p:return 'graph_analytics_documentation'
    if p.endswith(('hash_functions.md','functions-all.md')):return 'function_documentation'
    if any(x in p for x in ['mysql','migration','cutover','fallback']):return 'migration_cutover_fallback_documentation'
    if p.endswith(('instance-configurations.md','region-types.md','replication.md','geo-partitioning.md')):return 'replication_geo_documentation'
    if p.endswith('iam.md'):return 'iam_documentation'
    return 'catalog_format_or_editorial'
