package main

import (
	"cloud.google.com/go/spanner"
	database "cloud.google.com/go/spanner/admin/database/apiv1"
	dbpb "cloud.google.com/go/spanner/admin/database/apiv1/databasepb"
	sppb "cloud.google.com/go/spanner/apiv1/spannerpb"
	"context"
	"encoding/json"
	"errors"
	"github.com/apstndb/spanemuboost"
	"google.golang.org/api/iterator"
	"google.golang.org/grpc/status"
	"google.golang.org/protobuf/encoding/protojson"
	"os"
	"path/filepath"
	"regexp"
	"strings"
	"time"
)

var backend, phase, db string
var ctx = context.Background()

type step struct {
	Label      string   `json:"label"`
	Family     string   `json:"family"`
	Kind       string   `json:"kind"`
	SQL        string   `json:"sql"`
	Statements []string `json:"statements"`
	Requires   []string `json:"requires"`
	Table      string   `json:"table"`
	Key        []any    `json:"key"`
	After      string   `json:"after"`
	Profile    bool     `json:"profile"`
	Timeout    int      `json:"timeout_seconds"`
}
type input struct {
	Dialect string `json:"dialect"`
	Steps   []step `json:"steps"`
}

func scrub(v any) any {
	switch x := v.(type) {
	case string:
		if db != "" {
			x = strings.ReplaceAll(x, db, "<database>")
			x = strings.ReplaceAll(x, db[strings.LastIndex(db, "/")+1:], "<database>")
		}
		x = regexp.MustCompile(`projects/[^/\s"']+/instances/[^/\s"']+/databases/[^/\s"']+`).ReplaceAllString(x, "<database>")
		x = regexp.MustCompile(`projects/[^/\s"']+/instances/[^/\s"']+`).ReplaceAllString(x, "<instance>")
		return regexp.MustCompile(`requestID = "[^"]*"`).ReplaceAllString(x, `requestID = "<redacted>"`)
	case []any:
		out := make([]any, len(x))
		for i, v := range x {
			out[i] = scrub(v)
		}
		return out
	case []map[string]any:
		out := make([]map[string]any, len(x))
		for i, v := range x {
			out[i] = scrub(v).(map[string]any)
		}
		return out
	case map[string]any:
		out := map[string]any{}
		for k, v := range x {
			out[k] = scrub(v)
		}
		return out
	default:
		return v
	}
}
func emit(row map[string]any) {
	row["backend"] = backend
	row["phase"] = phase
	row["at"] = time.Now().UTC().Format(time.RFC3339Nano)
	if err := json.NewEncoder(os.Stdout).Encode(scrub(row)); err != nil {
		panic(err)
	}
}
func runPlan(rt spanemuboost.Runtime, path string) {
	phase = strings.TrimSuffix(filepath.Base(path), ".json")
	data, e := os.ReadFile(path)
	if e != nil {
		panic(e)
	}
	var plan input
	if e = json.Unmarshal(data, &plan); e != nil {
		panic(e)
	}
	dialect := dbpb.DatabaseDialect_GOOGLE_STANDARD_SQL
	if plan.Dialect == "pg" {
		dialect = dbpb.DatabaseDialect_POSTGRESQL
	}
	opts := []spanemuboost.Option{spanemuboost.WithRandomDatabaseID(), spanemuboost.ForceSchemaTeardown(), spanemuboost.WithDatabaseDialect(dialect)}
	if backend == "emulator" {
		opts = append(opts, spanemuboost.EnableAutoConfig())
	} else {
		opts = append(opts, spanemuboost.EnableDatabaseAutoConfigOnly())
	}
	bootstrap, cancel := context.WithTimeout(ctx, 3*time.Minute)
	clients, e := spanemuboost.OpenClients(bootstrap, rt, opts...)
	cancel()
	outcome("create-database", e)
	if e != nil {
		return
	}
	db = clients.DatabasePath()
	admin := clients.DatabaseClient
	client := clients.Client
	defer func() {
		outcome("drop-database", clients.Close())
		inspector, e := database.NewDatabaseAdminClient(ctx, rt.ClientOptions()...)
		if e != nil {
			outcome("cleanup-inspector", e)
			return
		}
		defer inspector.Close()
		cx, cancel := context.WithTimeout(ctx, time.Minute)
		defer cancel()
		_, e = inspector.GetDatabase(cx, &dbpb.GetDatabaseRequest{Name: db})
		outcome("database-absent", e)
	}()
	dbLeaf := db[strings.LastIndex(db, "/")+1:]
	quotedDB := "`" + dbLeaf + "`"
	if plan.Dialect == "pg" {
		quotedDB = "\"" + dbLeaf + "\""
	}
	codes := map[string]string{}
	rollbackSentinel := errors.New("owned fixture rollback control")
	for _, s := range plan.Steps {
		blocked := []string{}
		for _, label := range s.Requires {
			if codes[label] != "OK" {
				blocked = append(blocked, label)
			}
		}
		if len(blocked) > 0 {
			codes[s.Label] = "NOT_RUN"
			emit(map[string]any{"label": s.Label, "family": s.Family, "kind": s.Kind, "code": "NOT_RUN", "execution_status": "blocked_by_fixture_dependency", "requires": blocked})
			continue
		}
		started := time.Now().UTC()
		seconds := s.Timeout
		if seconds == 0 {
			seconds = 45
		}
		if s.Kind == "ddl" {
			seconds = 180
		}
		cx, cancel := context.WithTimeout(ctx, time.Duration(seconds)*time.Second)
		result := map[string]any{"label": s.Label, "family": s.Family, "kind": s.Kind, "sql": s.SQL, "statements": s.Statements, "started_at": started.Format(time.RFC3339Nano)}
		var e error
		switch s.Kind {
		case "ddl":
			statements := s.Statements
			if len(statements) == 0 {
				statements = []string{s.SQL}
			}
			statements = append([]string(nil), statements...)
			for i := range statements {
				statements[i] = strings.ReplaceAll(statements[i], "{{DATABASE}}", quotedDB)
			}
			op, err := admin.UpdateDatabaseDdl(cx, &dbpb.UpdateDatabaseDdlRequest{Database: db, Statements: statements})
			e = err
			if e == nil {
				e = op.Wait(cx)
			}
		case "query":
			ro := client.Single()
			mode := sppb.ExecuteSqlRequest_NORMAL
			if s.Profile {
				mode = sppb.ExecuteSqlRequest_PROFILE
			}
			it := ro.QueryWithOptions(cx, spanner.Statement{SQL: s.SQL}, spanner.QueryOptions{Mode: &mode})
			result["rows"], e = scan(it)
			if it.Metadata != nil {
				bytes, _ := protojson.Marshal(it.Metadata)
				result["metadata"] = json.RawMessage(bytes)
			}
			if it.QueryPlan != nil {
				bytes, _ := protojson.Marshal(it.QueryPlan)
				result["plan"] = json.RawMessage(bytes)
			}
			if it.QueryStats != nil {
				result["stats"] = it.QueryStats
			}
			it.Stop()
			ro.Close()
		case "dml":
			var counts []int64
			_, e = client.ReadWriteTransaction(cx, func(ctx context.Context, tx *spanner.ReadWriteTransaction) error {
				counts = nil
				for _, sql := range s.Statements {
					n, err := tx.Update(ctx, spanner.Statement{SQL: sql})
					if err != nil {
						return err
					}
					counts = append(counts, n)
				}
				return nil
			})
			result["modified_counts"] = counts
			result["transaction_committed"] = e == nil
		case "mutation-delete":
			key := make(spanner.Key, len(s.Key))
			for i, value := range s.Key {
				if number, ok := value.(float64); ok {
					value = int64(number)
				}
				key[i] = value
			}
			_, e = client.Apply(cx, []*spanner.Mutation{spanner.Delete(s.Table, key)})
			result["table"] = s.Table
			result["key"] = s.Key
		case "rollback":
			var inside []map[string]any
			_, e = client.ReadWriteTransaction(cx, func(ctx context.Context, tx *spanner.ReadWriteTransaction) error {
				_, err := tx.Update(ctx, spanner.Statement{SQL: s.SQL})
				if err != nil {
					return err
				}
				it := tx.Query(ctx, spanner.Statement{SQL: s.After})
				defer it.Stop()
				inside, err = scan(it)
				if err != nil {
					return err
				}
				return rollbackSentinel
			})
			result["rollback_expected"] = errors.Is(e, rollbackSentinel)
			result["rows_inside_rollback"] = inside
			result["transaction_committed"] = false
			if errors.Is(e, rollbackSentinel) {
				e = nil
			}
		case "ddl-export":
			response, err := admin.GetDatabaseDdl(cx, &dbpb.GetDatabaseDdlRequest{Database: db})
			e = err
			if e == nil {
				result["ddl"] = response.Statements
			}
		default:
			panic("unknown step kind")
		}
		cancel()
		codes[s.Label] = status.Code(e).String()
		result["code"] = codes[s.Label]
		result["error"] = ""
		if e != nil {
			result["error"] = e.Error()
		}
		result["completed_at"] = time.Now().UTC().Format(time.RFC3339Nano)
		emit(result)
	}
}
func outcome(label string, err error) {
	message := ""
	if err != nil {
		message = err.Error()
	}
	emit(map[string]any{"label": label, "code": status.Code(err).String(), "error": message})
}
func scan(it *spanner.RowIterator) ([]map[string]any, error) {
	rows := []map[string]any{}
	for {
		row, err := it.Next()
		if err == iterator.Done {
			return rows, nil
		}
		if err != nil {
			return rows, err
		}
		values := map[string]any{}
		for i, name := range row.ColumnNames() {
			var value spanner.GenericColumnValue
			if err := row.Column(i, &value); err != nil {
				return rows, err
			}
			values[name] = value.Value.AsInterface()
		}
		rows = append(rows, values)
	}
}

func main() {
	for _, key := range []string{"TEST_REAL_SPANNER_DATABASE", "SPANNER_PROJECT_ID", "SPANNER_INSTANCE_ID", "SPANNER_DATABASE_ID", "SPANNER_EMULATOR_HOST"} {
		if os.Getenv(key) != "" {
			panic("managed/emulator environment must be absent before owned runtime bootstrap")
		}
	}
	backend = os.Args[1]
	image := os.Args[2]
	phase = "bootstrap"
	kind := spanemuboost.BackendOmni
	if backend == "emulator" {
		kind = spanemuboost.BackendEmulator
	}
	rt, e := spanemuboost.Run(ctx, kind, spanemuboost.WithContainerImage(image), spanemuboost.DisableAutoConfig())
	outcome("runtime-start", e)
	if e != nil {
		os.Exit(1)
	}
	defer func() { phase = "cleanup"; outcome("runtime-cleanup", rt.Close()) }()
	platform, e := spanemuboost.RuntimePlatform(ctx, rt)
	if e != nil {
		panic(e)
	}
	emit(map[string]any{"label": "identity", "image": image, "platform": platform})
	for _, path := range os.Args[3:] {
		runPlan(rt, path)
	}
}
