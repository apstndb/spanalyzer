def retain(src, dst):
    dst.parent.mkdir(parents=True, exist_ok=True)
    if dst.exists():
        assert sha(src) == sha(dst), f'Refuse to replace retained observation: {dst}'
    else:
        shutil.copyfile(src, dst)
