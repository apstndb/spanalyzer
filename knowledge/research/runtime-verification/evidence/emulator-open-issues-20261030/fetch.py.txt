def comments(issue):
 if not issue['comments']:return issue['number'],[]
 cmd=['gh','api','--paginate','--slurp',f"repos/GoogleCloudPlatform/cloud-spanner-emulator/issues/{issue['number']}/comments?per_page=100"]
