var backend, phase string
var ctx = context.Background()
var c *spanner.Client
var a *database.DatabaseAdminClient
var db string
var dialect string
var q = "ManagedQueueProbe20261029r1"

func emit(v map[string]any) {
	v["at"] = time.Now().UTC().Format(time.RFC3339Nano)
	v["dialect"] = dialect
	v["backend"] = backend
	v["phase"] = phase
	json.NewEncoder(os.Stdout).Encode(scrub(v))
}
func clean(e error) string {
	if e == nil {
		return ""
	}
	s := e.Error()
	if db != "" {
		s = strings.ReplaceAll(s, db, "<database>")
	}
	for _, k := range []string{"SPANNER_PROJECT_ID", "SPANNER_INSTANCE_ID", "SPANNER_DATABASE_ID"} {
		if v := os.Getenv(k); v != "" {
			s = strings.ReplaceAll(s, v, "<redacted>")
		}
	}
	return s
}
func query(label, sql string, params map[string]any, stale bool, first bool) []map[string]any {
	start := time.Now()
	cx, cancel := context.WithTimeout(ctx, 35*time.Second)
	defer cancel()
	tx := c.Single()
	defer tx.Close()
	if stale {
		tx = tx.WithTimestampBound(spanner.ExactStaleness(time.Millisecond))
	}
	it := tx.Query(cx, spanner.Statement{SQL: sql, Params: params})
	defer it.Stop()
	rows := []map[string]any{}
	retained := []map[string]any{}
	types := map[string]string{}
	var err error
	for {
		var r *spanner.Row
		r, err = it.Next()
		if err != nil {
			break
		}
		row := map[string]any{}
		rr := map[string]any{}
		for i, n := range r.ColumnNames() {
			var g spanner.GenericColumnValue
			if err = r.Column(i, &g); err != nil {
				break
			}
			types[n] = g.Type.Code.String()
			v := g.Value.AsInterface()
			row[n] = v
			rr[n] = v
			if strings.Contains(n, "Token") && v != nil {
				rr[n] = fmt.Sprintf("sha256:%x", sha256.Sum256([]byte(fmt.Sprint(v))))
			}
		}
		if err != nil {
			break
		}
		rows = append(rows, row)
		retained = append(retained, rr)
		if first {
			if label == "receive-immediate" {
				if token, ok := row["SpannerLeaseToken"].(string); ok {
					leaseControls(token)
				}
			}
			break
		}
	}
	if err == iterator.Done {
		err = nil
	}
	emit(map[string]any{"kind": "query", "label": label, "sql": sql, "stale": stale, "stop_after_first": first, "rows": retained, "column_types": types, "error": clean(err), "code": status.Code(err).String(), "elapsed_ms": time.Since(start).Milliseconds()})
	return rows
}
func tx(label string, sqls []string, rollback bool) {
	var counts []int64
	cx, cancel := context.WithTimeout(ctx, 35*time.Second)
	defer cancel()
	_, err := c.ReadWriteTransaction(cx, func(cx context.Context, t *spanner.ReadWriteTransaction) error {
		counts = nil
		for _, sql := range sqls {
			n, e := t.Update(cx, spanner.Statement{SQL: sql})
			counts = append(counts, n)
			if e != nil {
				return e
			}
		}
		if rollback {
			return errors.New("intentional probe rollback")
		}
		return nil
	})
	emit(map[string]any{"kind": "transaction", "label": label, "sql": sqls, "rows_modified": counts, "intentional_rollback": rollback, "error": clean(err), "code": status.Code(err).String()})
}
func ddl(label, sql string) bool {
	cx, cancel := context.WithTimeout(ctx, 4*time.Minute)
	defer cancel()
	op, e := a.UpdateDatabaseDdl(cx, &dbpb.UpdateDatabaseDdlRequest{Database: db, Statements: []string{sql}})
	if e == nil {
		e = op.Wait(cx)
	}
	emit(map[string]any{"kind": "ddl", "label": label, "sql": sql, "error": clean(e), "code": status.Code(e).String()})
	return e == nil
}

func observe(label, sql string, params map[string]any, mode sppb.ExecuteSqlRequest_QueryMode) {
	cx, cancel := context.WithTimeout(ctx, 25*time.Second)
	defer cancel()
	ro := c.Single()
	defer ro.Close()
	it := ro.QueryWithOptions(cx, spanner.Statement{SQL: sql, Params: params}, spanner.QueryOptions{Mode: &mode})
	defer it.Stop()
	count := 0
	var e error
	start := time.Now()
	for {
		_, e = it.Next()
		if e != nil {
			break
		}
		count++
	}
	if e == iterator.Done {
		e = nil
	}
	v := map[string]any{"kind": "plan-probe", "label": label, "sql": sql, "params": params, "mode": mode.String(), "error": clean(e), "code": status.Code(e).String(), "rows": count, "elapsed_ms": time.Since(start).Milliseconds(), "stats": it.QueryStats}
	if it.QueryPlan != nil {
		b, _ := protojson.Marshal(it.QueryPlan)
		v["plan"] = json.RawMessage(b)
	}
	if it.Metadata != nil {
		b, _ := protojson.Marshal(it.Metadata)
		v["metadata"] = json.RawMessage(b)
	}
	emit(v)
}

func leaseControls(token string) {
	query("control-scalar-default", "SELECT * FROM RENEWLEASE_"+q+"(lease_tokens=>[@token])", map[string]any{"token": token}, false, false)
	query("control-scalar-off", "@{AUTO_PARAMETER='OFF'} SELECT * FROM RENEWLEASE_"+q+"(lease_tokens=>[@token])", map[string]any{"token": token}, false, false)
	rows := query("control-array-default", "SELECT * FROM RENEWLEASE_"+q+"(lease_tokens=>@tokens)", map[string]any{"tokens": []string{token}}, false, false)
	if len(rows) > 0 {
		if next, ok := rows[0]["SpannerNewLeaseToken"].(string); ok {
			token = next
		}
	}
	query("control-array-off", "@{AUTO_PARAMETER='OFF'} SELECT * FROM RENEWLEASE_"+q+"(lease_tokens=>@tokens)", map[string]any{"tokens": []string{token}}, false, false)
}

func scrub(v any) any {
	switch x := v.(type) {
	case string:
		if db != "" {
			x = strings.ReplaceAll(x, db, "<database>")
		}
		x = regexp.MustCompile(`projects/[^/\s"']+/instances/[^/\s"']+/databases/[^/\s"']+`).ReplaceAllString(x, "<database>")
		x = regexp.MustCompile(`projects/[^/\s"']+/instances/[^/\s"']+`).ReplaceAllString(x, "<instance>")
		return regexp.MustCompile(`requestID = "[^"]*"`).ReplaceAllString(x, `requestID = "<redacted>"`)
	case []any:
		out := make([]any, len(x))
		for i, v := range x {
			out[i] = scrub(v)
		}
		return out
	case []map[string]any:
		out := make([]map[string]any, len(x))
		for i, v := range x {
			out[i] = scrub(v).(map[string]any)
		}
		return out
	case map[string]any:
		out := map[string]any{}
		for k, v := range x {
			out[k] = scrub(v)
		}
		return out
	default:
		return v
	}
}

func lifecycle() {
	query("preflight", "SELECT 1 AS ready", nil, false, false)
	if !ddl("create", "CREATE QUEUE "+q+" (MessageId STRING(36) NOT NULL, Payload BYTES(MAX) NOT NULL) PRIMARY KEY(MessageId)") {
		return
	}
	defer func() {
		if ddl("cleanup", "DROP QUEUE "+q) {
			query("cleanup-confirmation", "SELECT COUNT(*) AS remaining FROM INFORMATION_SCHEMA.TABLES WHERE TABLE_NAME='"+q+"'", nil, false, false)
		}
	}()
	query("catalog-tables", "SELECT TABLE_NAME, TABLE_TYPE FROM INFORMATION_SCHEMA.TABLES WHERE TABLE_NAME = '"+q+"'", nil, false, false)
	query("catalog-columns", "SELECT COLUMN_NAME, SPANNER_TYPE, IS_NULLABLE, COLUMN_DEFAULT FROM INFORMATION_SCHEMA.COLUMNS WHERE TABLE_NAME = '"+q+"' ORDER BY ORDINAL_POSITION", nil, false, false)
	query("queue-metadata-surface", "SELECT TABLE_NAME, COLUMN_NAME, SPANNER_TYPE FROM INFORMATION_SCHEMA.COLUMNS WHERE TABLE_SCHEMA = 'INFORMATION_SCHEMA' AND (STRPOS(TABLE_NAME,'QUEUE') > 0 OR STRPOS(COLUMN_NAME,'QUEUE') > 0) ORDER BY TABLE_NAME, ORDINAL_POSITION", nil, false, false)
	query("queue-options", "SELECT TABLE_NAME, OPTION_NAME, OPTION_VALUE FROM INFORMATION_SCHEMA.TABLE_OPTIONS WHERE TABLE_NAME='"+q+"' ORDER BY OPTION_NAME", nil, false, false)
	query("queue-routines", "SELECT ROUTINE_NAME, ROUTINE_TYPE FROM INFORMATION_SCHEMA.ROUTINES WHERE STRPOS(ROUTINE_NAME,'"+q+"') > 0", nil, false, false)
	got, e := a.GetDatabaseDdl(ctx, &dbpb.GetDatabaseDdlRequest{Database: db})
	if e == nil {
		var matched []string
		for _, s := range got.Statements {
			if strings.Contains(s, q) {
				matched = append(matched, s)
			}
		}
		emit(map[string]any{"kind": "ddl-export", "statements": matched})
	} else {
		emit(map[string]any{"kind": "ddl-export", "error": clean(e)})
	}
	tx("enqueue-rollback", []string{"INSERT INTO " + q + " (MessageId,Payload,DeliverTime) VALUES ('rolled-back',b'rollback',CURRENT_TIMESTAMP())"}, true)
	query("after-rollback", "SELECT MessageId FROM "+q, nil, false, false)
	tx("enqueue-immediate-and-future", []string{"INSERT INTO " + q + " (MessageId,Payload,DeliverTime) VALUES ('immediate',b'hello',CURRENT_TIMESTAMP()),('future',b'later',TIMESTAMP_ADD(CURRENT_TIMESTAMP(), INTERVAL 1 HOUR))"}, false)
	query("ordinary-read", "SELECT MessageId, Payload, DeliverTime FROM "+q+" ORDER BY MessageId", nil, false, false)
	receive := "SELECT * FROM RECEIVE_" + q + "(max_duration=>'3s', max_batch_size=>1)"
	observe("receive-plan", receive, nil, sppb.ExecuteSqlRequest_PLAN)
	query("stale-ordinary", "SELECT MessageId FROM "+q+" ORDER BY MessageId", nil, true, false)
	// This focused retry omits stale receive so it cannot acquire the immediate message first.
	rows := query("receive-immediate", "SELECT * FROM RECEIVE_"+q+"(max_duration=>'20m', max_batch_size=>1)", nil, false, true)
	if len(rows) > 0 {
		if token, ok := rows[0]["SpannerLeaseToken"].(string); ok {
			query("renew-lease", "SELECT * FROM RENEWLEASE_"+q+"(lease_tokens=>[@token])", map[string]any{"token": token}, false, false)
		}
	}
	tx("ack-rollback", []string{"DELETE FROM " + q + " WHERE MessageId='immediate' ASSERT_ROWS_MODIFIED 1"}, true)
	query("after-ack-rollback", "SELECT MessageId FROM "+q+" ORDER BY MessageId", nil, false, false)
	tx("ack-commit", []string{"DELETE FROM " + q + " WHERE MessageId='immediate' ASSERT_ROWS_MODIFIED 1"}, false)
	tx("duplicate-ack", []string{"DELETE FROM " + q + " WHERE MessageId='immediate' ASSERT_ROWS_MODIFIED 1"}, false)
	query("future-not-received", receive, nil, false, false)
	query("future-still-stored", "SELECT MessageId FROM "+q, nil, false, false)
	query("invalid-renew-token", "SELECT * FROM RENEWLEASE_"+q+"(lease_tokens=>['not-a-token'])", nil, false, false)
	tx("enqueue-default-delivery", []string{"INSERT INTO " + q + " (MessageId,Payload) VALUES ('default-delivery',b'default')"}, false)
	query("default-delivery-stored", "SELECT MessageId, DeliverTime FROM "+q+" ORDER BY MessageId", nil, false, false)
}
func recreate() {
	q = "QueueRecreate20261029"
	query("preflight", "SELECT 1 AS ready", nil, false, false)
	owned := false
	defer func() {
		if owned && ddl("cleanup", "DROP QUEUE "+q) {
			query("cleanup-confirmation", "SELECT COUNT(*) AS remaining FROM INFORMATION_SCHEMA.TABLES WHERE TABLE_NAME='"+q+"'", nil, false, false)
		}
	}()
	for cycle := 1; cycle <= 2; cycle++ {
		label := fmt.Sprintf("cycle-%d", cycle)
		if !ddl(label+"-create", "CREATE QUEUE "+q+" (MessageId STRING(36) NOT NULL, Payload BYTES(MAX) NOT NULL) PRIMARY KEY(MessageId)") {
			return
		}
		owned = true
		id := fmt.Sprintf("generation-%d", cycle)
		tx(label+"-enqueue", []string{"INSERT INTO " + q + " (MessageId,Payload,DeliverTime) VALUES ('" + id + "',b'probe',CURRENT_TIMESTAMP())"}, false)
		rows := query(label+"-receive", "SELECT * FROM RECEIVE_"+q+"(max_duration=>'3s', max_batch_size=>1)", nil, false, true)
		expected := len(rows) == 1 && rows[0]["MessageId"] == id
		emit(map[string]any{"label": label + "-expected-generation", "matches": expected, "expected_message_id": id, "row_count": len(rows)})
		if !expected {
			return
		}
		if cycle == 1 {
			if !ddl(label+"-drop", "DROP QUEUE "+q) {
				return
			}
			owned = false
		}
	}
}

func event(label string, e error) {
	emit(map[string]any{"label": label, "code": status.Code(e).String(), "error": clean(e)})
}
func runStage(rt spanemuboost.Runtime, name string, probe func()) {
	phase = name
	dialect = "GOOGLE_STANDARD_SQL"
	opts := []spanemuboost.Option{spanemuboost.WithRandomDatabaseID(), spanemuboost.ForceSchemaTeardown()}
	if backend == "emulator" {
		opts = append(opts, spanemuboost.EnableAutoConfig())
	} else {
		opts = append(opts, spanemuboost.EnableDatabaseAutoConfigOnly())
	}
	clients, e := spanemuboost.OpenClients(ctx, rt, opts...)
	event("create-database", e)
	if e != nil {
		return
	}
	db = clients.DatabasePath()
	c = clients.Client
	a = clients.DatabaseClient
	defer func() {
		event("drop-database", clients.Close())
		inspector, e := database.NewDatabaseAdminClient(ctx, rt.ClientOptions()...)
		if e != nil {
			event("cleanup-inspector", e)
			return
		}
		defer inspector.Close()
		_, e = inspector.GetDatabase(ctx, &dbpb.GetDatabaseRequest{Name: db})
		event("database-absent", e)
	}()
	probe()
}
func main() {
	backend = os.Args[1]
	image := os.Args[2]
	phase = "bootstrap"
	kind := spanemuboost.BackendOmni
	if backend == "emulator" {
		kind = spanemuboost.BackendEmulator
	}
	rt, e := spanemuboost.Run(ctx, kind, spanemuboost.WithContainerImage(image), spanemuboost.DisableAutoConfig())
	event("runtime-start", e)
	if e != nil {
		os.Exit(1)
	}
	defer func() { phase = "cleanup"; event("runtime-cleanup", rt.Close()) }()
	platform, e := spanemuboost.RuntimePlatform(ctx, rt)
	if e != nil {
		panic(e)
	}
	emit(map[string]any{"label": "identity", "image": image, "platform": platform})
	runStage(rt, "queue-lifecycle-no-stale", lifecycle)
}
