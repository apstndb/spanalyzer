package main

import (
 "context"
 "encoding/json"
 "fmt"
 "os"
 "time"
 "cloud.google.com/go/spanner"
 database "cloud.google.com/go/spanner/admin/database/apiv1"
 dbpb "cloud.google.com/go/spanner/admin/database/apiv1/databasepb"
 "google.golang.org/api/option"
 "google.golang.org/grpc/status"
)

func emit(label string, err error, value any) {
 msg := ""; if err != nil { msg = err.Error() }
 json.NewEncoder(os.Stdout).Encode(map[string]any{"label":label,"code":status.Code(err).String(),"error":msg,"value":value,"at":time.Now().UTC()})
}
func main() {
 endpoint:=os.Getenv("OMNI_ENDPOINT")
 password,err:=os.ReadFile("password");if err!=nil { panic(err) }
 base:=spanner.ClientConfig{Type:spanner.OMNI,CaCertificateFile:"ca-api.crt"}
 mtls:=base;mtls.ClientCertificateFile="client.crt";mtls.ClientKeyFile="client.key"
 user:=base;user.Username="admin";user.Password=password
 bad:=user;bad.Password=[]byte("WrongProbe9!Password")
 db:="projects/default/instances/default/databases/r4secure"
 ctx,cancel:=context.WithTimeout(context.Background(),60*time.Second);defer cancel()
 a,err:=database.NewDatabaseAdminClientWithConfig(ctx,mtls,option.WithEndpoint(endpoint));if err!=nil {emit("admin-client",err,nil);return};defer a.Close()
 op,err:=a.CreateDatabase(ctx,&dbpb.CreateDatabaseRequest{Parent:"projects/default/instances/default",CreateStatement:"CREATE DATABASE r4secure"})
 if err==nil {_,err=op.Wait(ctx)};emit("create-mtls",err,nil);if err!=nil{return}
 for _,v:=range []struct{label string;cfg spanner.ClientConfig}{{"mtls",mtls},{"password",user},{"bad-password",bad},{"no-credentials",base}} {
  cx,done:=context.WithTimeout(context.Background(),15*time.Second)
  c,e:=spanner.NewClientWithConfig(cx,db,v.cfg,option.WithEndpoint(endpoint))
  var n int64
  if e==nil {tx:=c.Single();it:=tx.Query(cx,spanner.Statement{SQL:"SELECT 1"});row,x:=it.Next();e=x;if e==nil{e=row.Columns(&n)};it.Stop();tx.Close();c.Close()}
  emit(v.label,e,n);done()
 }
 cx,done:=context.WithTimeout(context.Background(),30*time.Second);defer done()
 emit("drop-mtls",a.DropDatabase(cx,&dbpb.DropDatabaseRequest{Database:db}),nil)
 fmt.Fprintln(os.Stderr,"Security probe completed")
}
