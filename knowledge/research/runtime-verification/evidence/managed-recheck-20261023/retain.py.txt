def normalized(value):

def paired(before,after):
    left,right=defaultdict(list),defaultdict(list)
    for row in before:left[(row.get('label'),row.get('kind'),normalized(row.get('sql')))].append(row)
    for row in after:right[(row.get('label'),row.get('kind'),normalized(row.get('sql')))].append(row)
    changes=[];count=0
    for key,rows in left.items():
        for a,b in zip(rows,right.get(key,[])):
            count+=1
            if a.get('code')!=b.get('code'):changes.append({'label':b.get('label'),'before':a.get('code'),'after':b.get('code'),'sql':b.get('sql')})
    return {'baseline_records':len(before),'current_records':len(after),'comparable_records':count,'changed_codes':changes,'value_scope':'Not compared: workload, tokens, timestamps, sampled rows and plan costs can vary'}

def walk(value):
    if isinstance(value,dict):
        yield value
        for item in value.values():yield from walk(item)
    elif isinstance(value,list):
        for item in value:yield from walk(item)
