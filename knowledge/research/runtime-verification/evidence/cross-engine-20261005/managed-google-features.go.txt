var ctx = context.Background()
var c *spanner.Client
var a *database.DatabaseAdminClient
var db string
var q = "QueueProbe20260918F"

func emit(v map[string]any) {
	v["at"] = time.Now().UTC().Format(time.RFC3339Nano)
	json.NewEncoder(os.Stdout).Encode(v)
}
func clean(e error) string {
	if e == nil {
		return ""
	}
	s := e.Error()
	s = strings.ReplaceAll(s, db, "<database>")
	for _, k := range []string{"SPANNER_PROJECT_ID", "SPANNER_INSTANCE_ID", "SPANNER_DATABASE_ID"} {
		if v := os.Getenv(k); v != "" {
			s = strings.ReplaceAll(s, v, "<redacted>")
		}
	}
	return s
}
func query(label, sql string, params map[string]any, stale bool, first bool) []map[string]any {
	start := time.Now()
	cx, cancel := context.WithTimeout(ctx, 35*time.Second)
	defer cancel()
	tx := c.Single()
	defer tx.Close()
	if stale {
		tx = tx.WithTimestampBound(spanner.ExactStaleness(time.Minute))
	}
	it := tx.Query(cx, spanner.Statement{SQL: sql, Params: params})
	defer it.Stop()
	rows := []map[string]any{}
	retained := []map[string]any{}
	types := map[string]string{}
	var err error
	for {
		var r *spanner.Row
		r, err = it.Next()
		if err != nil {
			break
		}
		row := map[string]any{}
		rr := map[string]any{}
		for i, n := range r.ColumnNames() {
			var g spanner.GenericColumnValue
			if err = r.Column(i, &g); err != nil {
				break
			}
			types[n] = g.Type.Code.String()
			v := g.Value.AsInterface()
			row[n] = v
			rr[n] = v
			if strings.Contains(n, "Token") && v != nil {
				rr[n] = fmt.Sprintf("sha256:%x", sha256.Sum256([]byte(fmt.Sprint(v))))
			}
		}
		if err != nil {
			break
		}
		rows = append(rows, row)
		retained = append(retained, rr)
		if first {
			if label == "receive-immediate" {
				if token, ok := row["SpannerLeaseToken"].(string); ok {
					query("renew-while-stream-open", "SELECT * FROM RENEWLEASE_"+q+"(lease_tokens=>[@token])", map[string]any{"token": token}, false, false)
				}
			}
			break
		}
	}
	if err == iterator.Done {
		err = nil
	}
	emit(map[string]any{"kind": "query", "label": label, "sql": sql, "stale": stale, "stop_after_first": first, "rows": retained, "column_types": types, "error": clean(err), "code": status.Code(err).String(), "elapsed_ms": time.Since(start).Milliseconds()})
	return rows
}
func tx(label string, sqls []string, rollback bool) {
	var counts []int64
	_, err := c.ReadWriteTransaction(ctx, func(cx context.Context, t *spanner.ReadWriteTransaction) error {
		counts = nil
		for _, sql := range sqls {
			n, e := t.Update(cx, spanner.Statement{SQL: sql})
			counts = append(counts, n)
			if e != nil {
				return e
			}
		}
		if rollback {
			return errors.New("intentional probe rollback")
		}
		return nil
	})
	emit(map[string]any{"kind": "transaction", "label": label, "sql": sqls, "rows_modified": counts, "intentional_rollback": rollback, "error": clean(err), "code": status.Code(err).String()})
}
func ddl(label, sql string) bool {
	cx, cancel := context.WithTimeout(ctx, 4*time.Minute)
	defer cancel()
	op, e := a.UpdateDatabaseDdl(cx, &dbpb.UpdateDatabaseDdlRequest{Database: db, Statements: []string{sql}})
	if e == nil {
		e = op.Wait(cx)
	}
	emit(map[string]any{"kind": "ddl", "label": label, "sql": sql, "error": clean(e), "code": status.Code(e).String()})
	return e == nil
}

func observe(label, sql string, params map[string]any, mode sppb.ExecuteSqlRequest_QueryMode) {
	cx, cancel := context.WithTimeout(ctx, 25*time.Second)
	defer cancel()
	ro := c.Single()
	defer ro.Close()
	it := ro.QueryWithOptions(cx, spanner.Statement{SQL: sql, Params: params}, spanner.QueryOptions{Mode: &mode})
	defer it.Stop()
	count := 0
	var e error
	start := time.Now()
	for {
		_, e = it.Next()
		if e != nil {
			break
		}
		count++
	}
	if e == iterator.Done {
		e = nil
	}
	v := map[string]any{"kind": "plan-probe", "label": label, "sql": sql, "params": params, "mode": mode.String(), "error": clean(e), "code": status.Code(e).String(), "rows": count, "elapsed_ms": time.Since(start).Milliseconds(), "stats": it.QueryStats}
	if it.QueryPlan != nil {
		b, _ := protojson.Marshal(it.QueryPlan)
		v["plan"] = json.RawMessage(b)
	}
	if it.Metadata != nil {
		b, _ := protojson.Marshal(it.Metadata)
		v["metadata"] = json.RawMessage(b)
	}
	emit(v)
}
func main() {
	if os.Getenv("SPANNER_EMULATOR_HOST") != "" {
		panic("emulator override")
	}
	db = fmt.Sprintf("projects/%s/instances/%s/databases/%s", os.Getenv("SPANNER_PROJECT_ID"), os.Getenv("SPANNER_INSTANCE_ID"), os.Getenv("SPANNER_DATABASE_ID"))
	var e error
	a, e = database.NewDatabaseAdminClient(ctx)
	if e != nil {
		panic(clean(e))
	}
	defer a.Close()
	c, e = spanner.NewClient(ctx, db)
	if e != nil {
		panic(clean(e))
	}
	defer c.Close()
	query("preflight", "SELECT 1 AS ready", nil, false, false)
	t := "ReleaseProbe20261005b"
	view := t + "View"
	if !ddl("create-table", "CREATE TABLE "+t+"(Id INT64 NOT NULL, Name STRING(MAX)) PRIMARY KEY(Id)") {
		return
	}
	defer func() {
		ddl("drop-table", "DROP TABLE "+t)
		query("cleanup", "SELECT TABLE_NAME FROM INFORMATION_SCHEMA.TABLES WHERE STARTS_WITH(TABLE_NAME,'"+t+"')", nil, false, false)
	}()
	_, e = c.Apply(ctx, []*spanner.Mutation{spanner.Insert(t, []string{"Id", "Name"}, []any{int64(1), "Alice"}), spanner.Insert(t, []string{"Id", "Name"}, []any{int64(2), "Bob"})})
	emit(map[string]any{"label": "seed", "error": clean(e), "code": status.Code(e).String()})
	for _, mode := range []string{"DEFAULT", "AUTO", "ON", "OFF", "INVALID"} {
		for _, id := range []int{1, 2} {
			prefix := ""
			if mode != "DEFAULT" {
				prefix = "@{AUTO_PARAMETER='" + mode + "'} "
			}
			sql := fmt.Sprintf("%sSELECT Id, Name FROM %s WHERE Id=%d", prefix, t, id)
			observe("auto-parameter-"+mode, sql, nil, sppb.ExecuteSqlRequest_PROFILE)
			query("result-"+mode, sql, nil, false, false)
		}
	}
	if ddl("create-view", "CREATE VIEW "+view+" SQL SECURITY INVOKER AS SELECT a.Id, a.Name FROM "+t+" a JOIN "+t+" b ON a.Id=b.Id") {
		for i, opt := range []string{"", " OPTIONS(validate_element_key_uniqueness=true)", " OPTIONS(validate_element_key_uniqueness=false)"} {
			g := fmt.Sprintf("%sG%d", t, i)
			if ddl("graph-uniqueness", "CREATE PROPERTY GRAPH "+g+" NODE TABLES("+view+" KEY(Id))"+opt) {
				query("graph-match", "GRAPH "+g+" MATCH(n) RETURN COUNT(*) AS n", nil, false, false)
				query("graph-metadata", "SELECT * FROM INFORMATION_SCHEMA.PROPERTY_GRAPHS WHERE PROPERTY_GRAPH_NAME='"+g+"'", nil, false, false)
				ddl("drop-graph", "DROP PROPERTY GRAPH "+g)
			}
		}
		ddl("drop-view", "DROP VIEW "+view)
	}
	g := t + "Semantic"
	if ddl("graph-semantic", "CREATE PROPERTY GRAPH "+g+" NODE TABLES("+t+" DEFAULT LABEL OPTIONS(description='People',synonyms=['person']) PROPERTIES(Id,Name OPTIONS(description='Person name',synonyms=['full name'])))") {
		query("semantic-metadata", "SELECT * FROM INFORMATION_SCHEMA.PROPERTY_GRAPHS WHERE PROPERTY_GRAPH_NAME='"+g+"'", nil, false, false)
		r, e := a.GetDatabaseDdl(ctx, &dbpb.GetDatabaseDdlRequest{Database: db})
		ss := []string{}
		if e == nil {
			for _, x := range r.Statements {
				if strings.Contains(x, g) {
					ss = append(ss, x)
				}
			}
		}
		emit(map[string]any{"label": "semantic-export", "ddl": ss, "error": clean(e), "code": status.Code(e).String()})
		ddl("drop-semantic", "DROP PROPERTY GRAPH "+g)
	}
 for version:=1;version<=9;version++ {
  query("approx-version",fmt.Sprintf("@{OPTIMIZER_VERSION=%d} SELECT APPROX_COUNT_DISTINCT(Name) AS n FROM %s",version,t),nil,false,false)
 }
 volatileView:=t+"VolatileView"
 if ddl("volatile-view","CREATE VIEW "+volatileView+" SQL SECURITY INVOKER AS SELECT v.Id, GENERATE_UUID() AS score FROM "+t+" v") {
  for i,opt:=range []string{"",", allow_view_with_volatile_column=false",", allow_view_with_volatile_column=true"} {
   name:=fmt.Sprintf("%sVolatileGraph%d",t,i)
   if ddl("volatile-graph-"+fmt.Sprint(i),"CREATE PROPERTY GRAPH "+name+" NODE TABLES("+volatileView+" KEY(Id)) OPTIONS(validate_element_key_uniqueness=false"+opt+")") {
    query("volatile-read-"+fmt.Sprint(i),"GRAPH "+name+" MATCH(n) RETURN n.Id, n.score",nil,false,false)
    ddl("volatile-drop-"+fmt.Sprint(i),"DROP PROPERTY GRAPH "+name)
   }
  }
  ddl("volatile-view-drop","DROP VIEW "+volatileView)
 }

}
